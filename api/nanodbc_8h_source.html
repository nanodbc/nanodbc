517template <class T>
518std::unique_ptr<bool[]>
519split_optional(std::vector<std::optional<T>> const& values, std::vector<T>& column)
520{
521 auto nulls = std::make_unique<bool[]>(values.size());
522 column.reserve(values.size());
523 std::size_t i = 0;
524 for (auto const& value : values)
525 {
526 nulls[i++] = !value.has_value();
527 column.push_back(value ? *value : T());
528 }
529 return nulls;
530}
